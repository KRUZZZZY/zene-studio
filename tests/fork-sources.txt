# include/MidiOss.h, plugins/Sf2Player/fluidsynthshims.h, src/core/audio/AudioOss.cpp,

# 4e677cb6c6ab (`git cat-file -e 4e677cb6c6ab:<path>`). The fork point is a fixed commit,
# so the corrected command needs no remote state at all. Verify it is in the clone first:
#   git cat-file -e 4e677cb6c6ab^{commit} && echo ok
# (`git merge-base origin/master HEAD` returns the same commit — 4e677cb6c6ab — but only
# while origin has been fetched; the literal commit is the base that cannot drift.)
#

# hand-added, and the rest of the fork's own tooling is registered in
# tests/tools-sources.txt, which carries its own regeneration command and baselines.
